if (ANDROID)
    # The qm loader generated by ecm_create_qm_loader() looks for translations
    # in the APK assets (assets:/share/locale/...), but the install step that
    # would normally lay out share/locale never runs for APK builds. Stage the
    # compiled catalogs into the package source dir instead; androiddeployqt
    # copies its assets/ into the package on every run. (Staging directly into
    # android-build/assets would not work: androiddeployqt wipes it.)
    set(android_package_dir ${CMAKE_BINARY_DIR}/android-package)
    file(GLOB po_langs RELATIVE ${CMAKE_SOURCE_DIR}/po ${CMAKE_SOURCE_DIR}/po/*)
    set(qm_asset_files "")
    foreach(lang ${po_langs})
        if (EXISTS ${CMAKE_SOURCE_DIR}/po/${lang}/syndic.po)
            set(qm_file ${CMAKE_BINARY_DIR}/ECMPoQm/${lang}/syndic.qm)
            set(qm_asset ${android_package_dir}/assets/share/locale/${lang}/LC_MESSAGES/syndic.qm)
            add_custom_command(OUTPUT ${qm_asset}
                COMMAND ${CMAKE_COMMAND} -E copy_if_different ${qm_file} ${qm_asset}
                DEPENDS ${qm_file}
                )
            list(APPEND qm_asset_files ${qm_asset})
        endif()
    endforeach()

    # Refresh the static package files from android/ and collect the staged
    # catalogs. Hooked into the package targets created when the syndic target
    # was finalized; a dependency on the (often already up-to-date) syndic
    # target itself would not reliably order this before androiddeployqt.
    add_custom_target(android-package-source
        COMMAND ${CMAKE_COMMAND} -E copy_directory_if_different
            ${CMAKE_SOURCE_DIR}/android ${android_package_dir}
        DEPENDS ${qm_asset_files}
        )
    foreach(package_target syndic_make_apk syndic_make_aab)
        if (TARGET ${package_target})
            add_dependencies(${package_target} android-package-source)
        endif()
    endforeach()
endif()
